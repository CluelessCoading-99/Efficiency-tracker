    // ========== UK BANK HOLIDAYS (England & Wales) ==========
    function easterSunday(year) {
      const a = year % 19;
      const b = Math.floor(year / 100);
      const c = year % 100;
      const d = Math.floor(b / 4);
      const e = b % 4;
      const f = Math.floor((b + 8) / 25);
      const g = Math.floor((b - f + 1) / 3);
      const h = (19 * a + b - d - g + 15) % 30;
      const i = Math.floor(c / 4);
      const k = c % 4;
      const l = (32 + 2 * e + 2 * i - h - k) % 7;
      const m = Math.floor((a + 11 * h + 22 * l) / 451);
      const month = Math.floor((h + l - 7 * m + 114) / 31);
      const day = ((h + l - 7 * m + 114) % 31) + 1;
      return new Date(year, month - 1, day);
    }

    function getUKBankHolidays(year) {
      const holidays = [];

      let ny = new Date(year, 0, 1);
      if (ny.getDay() === 0) ny = new Date(year, 0, 2);
      else if (ny.getDay() === 6) ny = new Date(year, 0, 3);
      holidays.push(ny);

      const easter = easterSunday(year);
      const goodFriday = new Date(easter);
      goodFriday.setDate(easter.getDate() - 2);
      const easterMonday = new Date(easter);
      easterMonday.setDate(easter.getDate() + 1);
      holidays.push(goodFriday, easterMonday);

      let earlyMay = new Date(year, 4, 1);
      while (earlyMay.getDay() !== 1) earlyMay.setDate(earlyMay.getDate() + 1);
      holidays.push(earlyMay);

      let spring = new Date(year, 4, 31);
      while (spring.getDay() !== 1) spring.setDate(spring.getDate() - 1);
      holidays.push(spring);

      let summer = new Date(year, 7, 31);
      while (summer.getDay() !== 1) summer.setDate(summer.getDate() - 1);
      holidays.push(summer);

      let xmas = new Date(year, 11, 25);
      if (xmas.getDay() === 0) xmas = new Date(year, 11, 26);
      else if (xmas.getDay() === 6) xmas = new Date(year, 11, 27);
      holidays.push(xmas);

      let boxing = new Date(year, 11, 26);
      if (boxing.getDay() === 0) boxing = new Date(year, 11, 27);
      else if (boxing.getDay() === 6) boxing = new Date(year, 11, 28);
      if (boxing.getTime() === xmas.getTime()) {
        boxing = new Date(xmas);
        boxing.setDate(xmas.getDate() + 1);
      }
      holidays.push(boxing);

      return holidays.map(d => {
        const y = d.getFullYear();
        const m = String(d.getMonth() + 1).padStart(2, '0');
        const day = String(d.getDate()).padStart(2, '0');
        return y + '-' + m + '-' + day;
      });
    }

    function countBankHolidaysInMonth(year, month) {

        return d.getFullYear() === year && d.getMonth() === month && d.getDay() !== 0 && d.getDay() !== 6;
      }).length;
    }

    function getHighestYear() {
      const ys = (appData.years || []).map(Number).filter(n => !Number.isNaN(n));
      return ys.length ? Math.max(...ys) : new Date().getFullYear();
    }

    function updateAddYearButtonState() {
      const selectedYr = parseInt(yearSelect.value, 10);
      const highestYr = getHighestYear();
      const canAdd = selectedYr === highestYr && !appData.years.includes(highestYr + 1);
      addYearBtn.classList.toggle('locked', !canAdd);
      addYearBtn.title = canAdd ? 'Add year ' + (highestYr + 1) : 'Switch to ' + highestYr + ' to add next year';
    }

    window.requestAddNextYear = function() {
      if (isAddingYear) return;
      const selectedYr = parseInt(yearSelect.value, 10);
      const highestYr = getHighestYear();
      if (selectedYr !== highestYr) {
        showToast('Switch to ' + highestYr + ' first to add the next year', 'warning');
        return;
      }
      const nextYr = highestYr + 1;
      if (appData.years.includes(nextYr)) return;
      pendingNextYear = nextYr;
      document.getElementById('addYearSplashTitle').textContent = 'Add Year ' + nextYr + '?';
      document.getElementById('addYearSplashMessage').textContent = 'Add ' + nextYr + ' to your tracker?';
      document.getElementById('addYearSplash').classList.remove('hidden');
    };

    window.closeAddYearSplash = () => {
      pendingNextYear = null;
      document.getElementById('addYearSplash').classList.add('hidden');
    };

    window.confirmAddNextYear = function() {
      if (isAddingYear || pendingNextYear === null) return;
      isAddingYear = true;
      try {
        const highestYr = getHighestYear();
        const expected = highestYr + 1;
        if (pendingNextYear === expected && !appData.years.includes(expected)) {
          appData.years.push(expected);
          appData.years = Array.from(new Set(appData.years.map(Number))).sort((a, b) => a - b);
          saveData();
          renderYearOptions(expected);
          refreshUI();
          showToast('Year ' + expected + ' added', 'success');
        }
      } finally {
        pendingNextYear = null;
        isAddingYear = false;
        closeAddYearSplash();
      }
    };

    window.shiftMonth = function(delta) {
      let y = parseInt(yearSelect.value, 10);
      let m = parseInt(monthSelect.value, 10) + delta;
      if (m < 0) { m = 11; y--; }
      if (m > 11) { m = 0; y++; }

      const minYear = Math.min(...appData.years);
      const maxYear = Math.max(...appData.years);
      if (y < minYear) {
        showToast('Earliest configured year is ' + minYear, 'warning');
        return;
      }
      if (y > maxYear) {
        showToast('Use + on ' + maxYear + ' to add ' + y, 'warning');
        return;
      }

      monthSelect.value = String(m);
      yearSelect.value = String(y);
      selectedJobIds.clear();
      refreshUI();
    };

    window.jumpToToday = function() {
      const now = new Date();
      const y = now.getFullYear();
      const m = now.getMonth();

      if (!appData.years.includes(y)) {
        appData.years.push(y);
        appData.years = Array.from(new Set(appData.years.map(Number))).sort((a, b) => a - b);

      }
      monthSelect.value = String(m);
      selectedJobIds.clear();
      refreshUI();
      showToast('Jumped to current period', 'info');
    };

    function collectCurrentFormSelections() {

      categoryCheckboxesContainer.querySelectorAll('input[name="mainCat"]:checked').forEach(cb => {

        selections[catKey] = {};
        dynamicChecklistsContainer.querySelectorAll('input[name="sub_' + catKey + '"]:checked').forEach(sub => {

    function getWeekdaysInMonth(year, month) {
      let count = 0;
      const days = new Date(year, month + 1, 0).getDate();
      for (let d = 1; d <= days; d++) {
        const dow = new Date(year, month, d).getDay();
        if (dow !== 0 && dow !== 6) count++;
      }
      return count;
    }

    function getMonthKey(year, month) { return year + '-' + month; }

    function getMonthConfig(year, month) {
      const key = getMonthKey(year, month);
      if (!appData.monthSettings[key]) {
        const prevM = month === 0 ? 11 : month - 1;
        const prevY = month === 0 ? year - 1 : year;
        const prev = appData.monthSettings[getMonthKey(prevY, prevM)];
        appData.monthSettings[key] = {
          dailyTarget: prev ? prev.dailyTarget : (appData.defaultDailyTarget || 8.0),
          extraDays: 0,
          holidayDays: 0,
          sickDays: 0
        };
      }
      return appData.monthSettings[key];
    }

    function getMonthTotals(year, month) {
      const config = getMonthConfig(year, month);
      const baseWeekdays = getWeekdaysInMonth(year, month);
      const bankHolidays = countBankHolidaysInMonth(year, month);

      const effectiveDays = Math.max(0,
        baseWeekdays
        + (parseFloat(config.extraDays) || 0)
        - (parseFloat(config.holidayDays) || 0)
        - bankHolidays
      );

      const targetHours = effectiveDays * (parseFloat(config.dailyTarget) || 8.0);

      const monthJobs = (appData.jobs || []).filter(job => {
        if (!job.date) return false;
        const d = new Date(job.date + 'T00:00:00');
        return !Number.isNaN(d.getTime()) && d.getFullYear() === year && d.getMonth() === month;
      });

      let totalReq = 0, totalGiven = 0;
      monthJobs.forEach(j => {
        totalReq += parseFloat(j.reqHrs) || 0;
        if (j.givenHrs !== null && j.givenHrs !== '') totalGiven += parseFloat(j.givenHrs) || 0;
      });

      return { totalReq, totalGiven, targetHours, baseWeekdays, effectiveDays, config, monthJobs, bankHolidays };
    }

    function refreshUI() {
      updateAddYearButtonState();

      const now = new Date();
      const year = parseInt(yearSelect.value, 10) || now.getFullYear();
      const month = parseInt(monthSelect.value, 10);
      const m = Number.isNaN(month) ? now.getMonth() : month;

      const { totalReq, totalGiven, targetHours, baseWeekdays, effectiveDays, config, monthJobs, bankHolidays } = getMonthTotals(year, m);

      let prevY = year, prevM = m - 1;
      if (prevM < 0) { prevM = 11; prevY--; }
      const prev = getMonthTotals(prevY, prevM);

      const reqPct = targetHours > 0 ? (totalReq / targetHours) * 100 : 0;
      const givenPct = targetHours > 0 ? (totalGiven / targetHours) * 100 : 0;
      const convPct = totalReq > 0 ? (totalGiven / totalReq) * 100 : 0;
      const remaining = Math.max(0, targetHours - totalGiven);

      const prevReqPct = prev.targetHours > 0 ? (prev.totalReq / prev.targetHours) * 100 : 0;
      const prevGivenPct = prev.targetHours > 0 ? (prev.totalGiven / prev.targetHours) * 100 : 0;

      function formatDelta(curr, prevVal) {
        if (prevVal === 0 && curr === 0) return { text: '—', cls: 'flat' };
        const diff = curr - prevVal;
        if (Math.abs(diff) < 0.15) return { text: '≈ same as last month', cls: 'flat' };
        const sign = diff > 0 ? '+' : '';
        return { text: sign + diff.toFixed(1) + '% vs last month', cls: diff > 0 ? 'up' : 'down' };
      }

      const reqDelta = formatDelta(reqPct, prevReqPct);
      const givenDelta = formatDelta(givenPct, prevGivenPct);

      document.getElementById('reqPercent').textContent = reqPct.toFixed(1) + '%';
      document.getElementById('reqDelta').textContent = reqDelta.text;
      document.getElementById('reqDelta').className = 'kpi-delta ' + reqDelta.cls;
      document.getElementById('reqBar').style.width = Math.min(100, reqPct) + '%';
      document.getElementById('reqSubtext').textContent = totalReq.toFixed(1) + ' / ' + targetHours.toFixed(1) + ' hrs';

      document.getElementById('givenPercent').textContent = givenPct.toFixed(1) + '%';
      document.getElementById('givenDelta').textContent = givenDelta.text;
      document.getElementById('givenDelta').className = 'kpi-delta ' + givenDelta.cls;
      document.getElementById('givenBar').style.width = Math.min(100, givenPct) + '%';
      document.getElementById('givenSubtext').textContent = totalGiven.toFixed(1) + ' / ' + targetHours.toFixed(1) + ' hrs';

      document.getElementById('chartTargetLabel').textContent = 'Target: ' + targetHours.toFixed(1) + 'h';
      document.getElementById('conversionBadge').textContent = 'Conv: ' + convPct.toFixed(0) + '%';
      document.getElementById('baseDaysDisplay').textContent = baseWeekdays + 'd';
      const netDaysSub = document.getElementById('netDaysSub');
      netDaysSub.textContent = effectiveDays !== baseWeekdays ? (effectiveDays + 'd net') : 'All working';
      document.getElementById('adjTargetDisplay').textContent = targetHours.toFixed(1) + 'h';
      document.getElementById('remainingHrsDisplay').textContent = remaining.toFixed(1) + 'h';

      const remainingCell = document.getElementById('remainingCell');
      if (remainingCell) {
        remainingCell.classList.toggle('complete', remaining <= 0 && targetHours > 0);
      }

      const noteEl = document.getElementById('bankHolidayNote');
      if (bankHolidays > 0) {
        noteEl.textContent = bankHolidays + ' UK Bank Holiday' + (bankHolidays > 1 ? 's' : '') + ' deducted (' + baseWeekdays + ' weekdays → ' + effectiveDays + ' net working days)';
        noteEl.classList.remove('hidden');
      } else {
        noteEl.classList.add('hidden');
      }

      dailyTargetInput.value = config.dailyTarget;
      extraDaysInput.value = config.extraDays;
      holidayDaysInput.value = config.holidayDays;
      sickDaysInput.value = config.sickDays;

      renderJobList(monthJobs, year, m);
    }

    function renderJobList(monthJobs, selectedYear, selectedMonth) {
      const filter = filterSelect.value;
      const sort = sortSelect.value;
      const rawQuery = searchInput.value.trim().toLowerCase();
      const compactQuery = rawQuery.replace(/\s+/g, '');
      const isSearching = compactQuery.length > 0;

      clearSearchBtn.classList.toggle('hidden', !isSearching);
      const todayStr = getLocalDateISO();

      const candidateJobs = isSearching ? (appData.jobs || []) : monthJobs;

      const filtered = candidateJobs.filter(job => {
        const isPending = job.givenHrs === null || job.givenHrs === '';
        const req = parseFloat(job.reqHrs) || 0;
        const given = parseFloat(job.givenHrs) || 0;
        const diff = Math.abs(req - given);
        const isFlagged = !isPending && diff > 0.20;

        if (!isSearching) {
          if (filter === 'today' && job.date !== todayStr) return false;
          if (filter === 'pending' && !isPending) return false;
          if (filter === 'flagged' && !isFlagged) return false;
        }

        if (isSearching) {
          const wipMatch = (job.wip || '').toLowerCase().replace(/\s+/g, '').includes(compactQuery);
          const regMatch = (job.reg || '').toLowerCase().replace(/\s+/g, '').includes(compactQuery);
          const descMatch = (job.desc || '').toLowerCase().includes(rawQuery);
          const commentMatch = (job.comment || '').toLowerCase().includes(rawQuery);
          return wipMatch || regMatch || descMatch || commentMatch;
        }
        return true;
      });

      filtered.sort((a, b) => {
        if (sort === 'date-desc') return (b.date || '').localeCompare(a.date || '');
        if (sort === 'date-asc') return (a.date || '').localeCompare(b.date || '');
        if (sort === 'wip') return String(a.wip).localeCompare(String(b.wip), undefined, { numeric: true });
        if (sort === 'reg') return String(a.reg || '').localeCompare(String(b.reg || ''));
        if (sort === 'diff-desc') {
          const da = Math.abs((parseFloat(a.givenHrs) || 0) - (parseFloat(a.reqHrs) || 0));
          const db = Math.abs((parseFloat(b.givenHrs) || 0) - (parseFloat(b.reqHrs) || 0));
          return db - da;
        }
        return 0;
      });

      currentRenderedJobs = filtered;

      if (isSearching) {
        searchScopeNote.classList.remove('hidden');
        searchScopeText.textContent = 'Showing ' + filtered.length + ' match(es) across all logged entries';
      } else {
        searchScopeNote.classList.add('hidden');
      }

      const visibleIds = new Set(filtered.map(j => String(j.id)));
      Array.from(selectedJobIds).forEach(id => { if (!visibleIds.has(String(id))) selectedJobIds.delete(id); });

      jobList.innerHTML = '';

      if (filtered.length === 0) {
        jobList.innerHTML = '<div class="empty-state"><strong>No jobs found</strong>Try changing the filter or search.</div>';
        updateBatchBar();
        return;
      }

      filtered.forEach(job => {
        const jobIdStr = String(job.id);
        const isPending = job.givenHrs === null || job.givenHrs === '';
        const req = parseFloat(job.reqHrs) || 0;
        const given = isPending ? 0 : parseFloat(job.givenHrs) || 0;
        const signed = given - req;
        const abs = Math.abs(signed);

        let flagClass, flagBadge;
        if (isPending) {
          flagClass = 'pending';
          flagBadge = '<span class="flag-badge badge-amber">Pending</span>';
        } else if (abs === 0) {
          flagClass = 'flag-green';
          flagBadge = '<span class="flag-badge badge-green">Exact Match</span>';
        } else if (signed > 0) {
          if (abs > 0.50) {
            flagClass = 'flag-blue';
            flagBadge = '<span class="flag-badge badge-blue">Over +' + abs.toFixed(2) + 'h</span>';
          } else {
            flagClass = 'flag-blue-soft';
            flagBadge = '<span class="flag-badge badge-blue-soft">Over +' + abs.toFixed(2) + 'h</span>';
          }
        } else {
          if (abs > 0.50) {
            flagClass = 'flag-red';
            flagBadge = '<span class="flag-badge badge-red">Under -' + abs.toFixed(2) + 'h</span>';
          } else if (abs > 0.20) {
            flagClass = 'flag-amber';
            flagBadge = '<span class="flag-badge badge-amber">Under -' + abs.toFixed(2) + 'h</span>';
          } else {
            flagClass = 'flag-green';
            flagBadge = '<span class="flag-badge badge-green">Close ≤0.2h</span>';
          }
        }

        const givenDisplay = isPending
          ? '<span style="color:var(--amber)">Pending</span>'
          : '<strong style="color:var(--emerald-bright)">' + parseFloat(job.givenHrs).toFixed(1) + 'h</strong>';

        let tagsHTML = '';
        if (job.categories) {
          const activeTags = [];
          Object.entries(job.categories).forEach(([catKey, itemsObj]) => {
            const catName = AVAILABLE_CATEGORIES[catKey] ? AVAILABLE_CATEGORIES[catKey].label : catKey;
            const checkedItems = Object.entries(itemsObj)
              .filter(([name, checked]) => checked && name !== 'isGWagon')
              .map(([name]) => name);
            const extra = (catKey === 'brakes' && itemsObj['isGWagon']) ? ' [G Wagon]' : '';
            if (checkedItems.length) {
              activeTags.push('<span class="tag">✓ ' + escapeHTML(catName + extra + ': ' + checkedItems.join(', ')) + '</span>');
            } else {
              activeTags.push('<span class="tag">✓ ' + escapeHTML(catName + extra) + '</span>');
            }
          });
          if (activeTags.length) tagsHTML = '<div class="job-checklist-tags">' + activeTags.join('') + '</div>';
        }

        let commentHTML = '';
        if (job.comment && job.comment.trim()) {
          commentHTML = '<div class="speech-bubble" onclick="editJobComment(\'' + escapeHTML(jobIdStr) + '\')" title="Edit comment">💬 ' + escapeHTML(job.comment) + '</div>';
        }

        const isSelected = selectedJobIds.has(jobIdStr);
        const isChecked = isSelected ? 'checked' : '';
        const quickGivenBtn = isPending
          ? '<button type="button" class="btn-quick-given" onclick="quickSetGiven(\'' + escapeHTML(jobIdStr) + '\')">Set Given</button>'
          : '';

        const jd = new Date((job.date || '') + 'T00:00:00');
        const isOutsideSelectedPeriod = isSearching && !Number.isNaN(jd.getTime()) &&
          (jd.getFullYear() !== selectedYear || jd.getMonth() !== selectedMonth);
        const periodTag = isOutsideSelectedPeriod
          ? ' <span class="flag-badge badge-blue-soft" style="font-size:0.64rem;">Other Month</span>'
          : '';

        const item = document.createElement('div');
        item.className = 'job-item ' + flagClass + (isSelected ? ' is-selected-for-print' : '');
        item.dataset.jobId = jobIdStr;
        item.innerHTML =
          '<div class="job-top">' +
            '<div class="job-title-group">' +
              '<input type="checkbox" class="job-checkbox" value="' + escapeHTML(jobIdStr) + '" ' + isChecked + ' onchange="toggleJobSelection(\'' + escapeHTML(jobIdStr) + '\', this)" style="width:17px;height:17px;accent-color:var(--cyan);cursor:pointer;" />' +
              '<span class="wip-pill" onclick="editJobComment(\'' + escapeHTML(jobIdStr) + '\')" title="Add/edit comment">WIP #' + escapeHTML(job.wip) + '</span>' +
              '<a class="reg-link" onclick="showVehicleHistory(\'' + escapeHTML((job.reg || '').replace(/'/g, "\\'")) + '\')" title="View vehicle history">' + escapeHTML(job.reg || '') + '</a>' +
              flagBadge + periodTag +
            '</div>' +
            '<span style="font-size:0.78rem;color:var(--muted);white-space:nowrap;">' + escapeHTML(job.date) + '</span>' +
          '</div>' +
          '<div class="job-desc">' + escapeHTML(job.desc || '') + '</div>' +
          commentHTML + tagsHTML +
          '<div class="job-metrics">' +
            '<div>Shouted / Called: <strong>' + req.toFixed(1) + 'h</strong> · Paid / Given: ' + givenDisplay + '</div>' +
            '<div class="job-actions">' +

    window.printJobs = function() {
      const hasSelection = selectedJobIds.size > 0;
      const jobsToPrint = hasSelection
        ? currentRenderedJobs.filter(j => selectedJobIds.has(String(j.id)))
        : currentRenderedJobs;

      if (jobsToPrint.length === 0) {
        showToast('No jobs visible to print', 'warning');
        return;
      }

      let totalShouted = 0;
      let totalPaid = 0;
      let pendingCount = 0;

      jobsToPrint.forEach(j => {
        totalShouted += parseFloat(j.reqHrs) || 0;
        if (j.givenHrs === null || j.givenHrs === '') {
          pendingCount++;
        } else {
          totalPaid += parseFloat(j.givenHrs) || 0;
        }
      });

      const netDiff = totalPaid - totalShouted;
      const diffSign = netDiff > 0 ? '+' : '';
      const tech = getActiveTechnician();
      const monthName = monthSelect.options[monthSelect.selectedIndex]?.textContent || '';
      const yearText = yearSelect.value || '';
      const filterLabel = hasSelection
        ? 'Selected Jobs (' + jobsToPrint.length + ')'
        : (filterSelect.options[filterSelect.selectedIndex]?.textContent || 'Job List') + ' (' + jobsToPrint.length + ' jobs)';

      const printHeader = document.getElementById('printSheetHeader');
      printHeader.innerHTML =
        '<div class="print-sheet-title">Workshop Job Sheet — ' + escapeHTML(filterLabel) + '</div>' +
        '<div class="print-sheet-meta">Technician: <strong>' + escapeHTML(tech.name) + '</strong> &bull; Period: <strong>' + escapeHTML(monthName + ' ' + yearText) + '</strong> &bull; Printed: ' + escapeHTML(getLocalDateISO()) + '</div>' +
        '<div class="print-sheet-totals">' +
          '<span><strong>Total Shouted:</strong> ' + totalShouted.toFixed(1) + 'h</span>' +
          '<span><strong>Total Paid:</strong> ' + totalPaid.toFixed(1) + 'h</span>' +
          '<span><strong>Net Variance:</strong> ' + diffSign + netDiff.toFixed(2) + 'h</span>' +
          (pendingCount > 0 ? '<span><strong>Pending:</strong> ' + pendingCount + '</span>' : '') +
        '</div>';

      if (hasSelection) {
        document.body.classList.add('print-selected-only');
      } else {
        document.body.classList.remove('print-selected-only');
      }

      window.print();
    };

    document.getElementById('importCsvBtn').addEventListener('click', () => {
      const file = document.getElementById('csvFileInput').files[0];
      if (!file) {
        showToast('Please select a CSV file first', 'warning');
        return;
      }

      const activeTech = getActiveTechnician();
      const targetMatch = (activeTech.match || activeTech.name).toLowerCase().trim();

      const reader = new FileReader();
      reader.onload = function(e) {
        const text = (e.target.result || '').replace(/^\uFEFF/, '');
        const lines = text.split(/\r\n|\n/);
        if (lines.length < 2) {
          showToast('CSV appears empty', 'error');
          return;
        }

        let updatedCount = 0;
        let seenCsvWips = new Set();

        function normalizeCsvDate(raw) {
          const s = String(raw || '').trim();
          if (!s) return null;
          if (/^\d{4}-\d{2}-\d{2}/.test(s)) return s.substring(0, 10);
          const dmy = s.match(/^(\d{1,2})[\/-](\d{1,2})[\/-](\d{4})$/);
          if (dmy) return dmy[3] + '-' + dmy[2].padStart(2, '0') + '-' + dmy[1].padStart(2, '0');
          return null;
        }

        function parseHourVal(str) {
          if (!str) return null;
          const cleaned = String(str).replace(/\u00a0/g, ' ').toLowerCase().replace(/hrs?/g, '').trim().replace(',', '.');
          const val = parseFloat(cleaned);
          return isNaN(val) ? null : val;
        }

        for (let i = 1; i < lines.length; i++) {
          const line = lines[i].trim();
          if (!line) continue;

          const commaCount = (line.match(/,(?=(?:(?:[^"]*"){2})*[^"]*$)/g) || []).length;
          const semiCount = (line.match(/;(?=(?:(?:[^"]*"){2})*[^"]*$)/g) || []).length;
          const delimiter = (semiCount > commaCount && semiCount >= 4) ? ';' : ',';
          const splitRegex = delimiter === ';'
            ? /;(?=(?:(?:[^"]*"){2})*[^"]*$)/
            : /,(?=(?:(?:[^"]*"){2})*[^"]*$)/;

          const parts = line.split(splitRegex).map(p => p.trim().replace(/^["']|["']$/g, ''));

          if (parts.length >= 2) {
            let techName = '', fileWip = '', fileDate = '', reqStr = '', givenStr = '';

            if (parts[0] && parts[0].toLowerCase().includes(targetMatch)) {
              techName = parts[0]; fileWip = parts[1] || ''; fileDate = parts[2] || ''; reqStr = parts[3] || ''; givenStr = parts[4] || '';
            } else if (parts[1] && parts[1].toLowerCase().includes(targetMatch)) {
              techName = parts[1]; fileWip = parts[2] || ''; fileDate = parts[3] || ''; reqStr = parts[4] || ''; givenStr = parts[5] || '';
            } else {
              techName = parts[0] || parts[1] || '';
            }

            if (!techName.toLowerCase().includes(targetMatch)) continue;

            const wipKey = String(fileWip || '').replace(/\s*hrs?\s*$/i, '').trim();
            const isNumericWip = /^\d+$/.test(wipKey);
            const normDate = normalizeCsvDate(fileDate);
            const fileReq = parseHourVal(reqStr);
            const fileGiven = parseHourVal(givenStr);

            if (wipKey && wipKey.toLowerCase() !== 'wip' && isNumericWip) {
              const matchedJob = appData.jobs.find(j => String(j.wip).trim() === wipKey);
              if (matchedJob) {
                if (!seenCsvWips.has(wipKey)) {
                  if (fileGiven !== null) matchedJob.givenHrs = fileGiven;
                  if (fileReq !== null && !isNaN(fileReq)) matchedJob.reqHrs = fileReq;
                  seenCsvWips.add(wipKey);
                } else {
                  if (fileGiven !== null) matchedJob.givenHrs = Math.round(((matchedJob.givenHrs || 0) + fileGiven) * 100) / 100;
                  if (fileReq !== null && !isNaN(fileReq)) matchedJob.reqHrs = Math.round(((matchedJob.reqHrs || 0) + fileReq) * 100) / 100;
                }
                updatedCount++;
              } else {
                appData.jobs.push({
                  id: generateId(),
                  date: normDate || getLocalDateISO(),
                  wip: wipKey,
                  reg: 'UNKNOWN',
                  categories: {},
                  desc: 'Imported from CSV',
                  reqHrs: fileReq !== null ? fileReq : 0.0,
                  givenHrs: fileGiven !== null ? fileGiven : null,
                  comment: '',
                  externalSyncId: null
                });
                seenCsvWips.add(wipKey);
                updatedCount++;
              }
            }
          }
        }

        saveData();
        selectedJobIds.clear();
        refreshUI();
        showToast('CSV processed — ' + updatedCount + ' row(s) updated/created', 'success');
        document.getElementById('csvFileInput').value = '';
      };
      reader.readAsText(file);
    });
